Add these 2 tools to your VAPI agent. Tool endpoint URL:

Tool 1: getInsuranceTypes
Call this at start of conversation — returns list of insurance types available.
{
  "name": "getInsuranceTypes",
  "description": "Get the list of insurance types available for this vendor",
  "parameters": { "type": "object", "properties": {} }
}
Tool 2: sendInsuranceLink
Call after collecting customer info — sends SMS with form link and saves enquiry.
{
  "name": "sendInsuranceLink",
  "description": "Send the insurance form link to customer via SMS and save enquiry",
  "parameters": {
    "type": "object",
    "properties": {
      "customer_name":  { "type": "string", "description": "Full name of the customer" },
      "customer_phone": { "type": "string", "description": "Customer phone number with country code" },
      "customer_email": { "type": "string", "description": "Customer email address" },
      "insurance_type": { "type": "string", "description": "Insurance type key from getInsuranceTypes" }
    },
    "required": ["customer_phone", "insurance_type"]
  }
}
